    } catch {
      return url;
    }
  }

  function getImage(item) {
    const img = item.querySelector('img[data-src], img[data-lazy-src], img');
    if (!img) return '';
    return absolute(
      img.getAttribute('data-src') ||
      img.getAttribute('data-lazy-src') ||
      img.currentSrc ||
      img.src ||
      ''
    );
  }

  function cleanText(value) {
    return (value || '').replace(/\\s+/g, ' ').trim();
  }

  function isHome() {
    const path = location.pathname.replace(/\\/+$/, '') || '/';
    return path === '/';
  }

  function isListing() {
    const p = location.pathname;
    return p.includes('/top-games') ||
           p.includes('/updated-games') ||
           p.includes('/games-list') ||
           p.includes('/collections') ||
           p.includes('/category/');
  }

  function isGamePage() {
    return !!document.querySelector('h1.post-title.entry-title, #the-post.entry-content');
  }

  function addClasses() {
    document.documentElement.id = 'sr-pro-redesign-html';
    document.body.classList.add('sr-redesign-active');
  }

  function buildMobileControls() {
    const nav = document.querySelector('#main-nav .main-menu-wrapper');
    if (!nav || nav.querySelector('.sr-mobile-bar')) return;

    const bar = document.createElement('div');
    bar.className = 'sr-mobile-bar';

    const search = document.createElement('button');
    search.type = 'button';
    search.title = 'Search';
    search.textContent = '⌕';
    search.addEventListener('click', openSearch);

    const menu = document.createElement('button');
    menu.type = 'button';
    menu.title = 'Menu';
    menu.textContent = '☰';
    menu.addEventListener('click', () => {
      const menuEl = document.querySelector('#main-nav-menu');
      if (!menuEl) return;
      const visible = menuEl.dataset.srMobileOpen === '1';
      menuEl.dataset.srMobileOpen = visible ? '0' : '1';
      menuEl.style.display = visible ? '' : 'block';
      menuEl.style.position = 'absolute';
      menuEl.style.left = '12px';
      menuEl.style.right = '12px';
      menuEl.style.top = '68px';
      menuEl.style.padding = '8px';
      menuEl.style.background = '#111019';
      menuEl.style.border = '1px solid rgba(255,255,255,.12)';
      menuEl.style.borderRadius = '12px';
      menuEl.style.boxShadow = '0 25px 70px rgba(0,0,0,.45)';
      if (!visible) {
        menuEl.querySelector('.menu').style.display = 'block';
        menuEl.querySelectorAll('.menu > li > a').forEach(a => {
          a.style.width = '100%';
          a.style.justifyContent = 'flex-start';
        });
      }
    });

    bar.append(search, menu);
    nav.appendChild(bar);
  }

  function buildSearch() {
    if (document.querySelector('.sr-search')) return;

    const overlay = document.createElement('div');
    overlay.className = 'sr-search';
    overlay.innerHTML = `
      <div class="sr-search-box" role="dialog" aria-label="Search SteamRIP">
        <div class="sr-search-head">
          <span style="color:#a855f7;font-size:20px">⌕</span>
          <input class="sr-search-input" type="search" placeholder="Search games..." autocomplete="off">
        </div>
        <div class="sr-search-results"></div>
      </div>
    `;

    overlay.addEventListener('click', e => {
      if (e.target === overlay) closeSearch();
    });

    document.body.appendChild(overlay);

    const input = overlay.querySelector('.sr-search-input');
    const results = overlay.querySelector('.sr-search-results');

    input.addEventListener('input', () => {
      const query = cleanText(input.value).toLowerCase();
      results.innerHTML = '';
      if (query.length < 2) return;

      const cards = [...document.querySelectorAll('.post-item')];
      const matches = cards
        .map(item => {
          const link = item.querySelector('h2 a, h2.the-post-title, a.post-thumb, .post-title a');
          const a = item.querySelector('a[href]');
          const title = cleanText(
            item.querySelector('.the-post-title, .post-title')?.textContent ||
            link?.textContent ||
            ''
          );
          return {
            title,
            href: a ? absolute(a.getAttribute('href')) : '',
            image: getImage(item)
          };
        })
        .filter(x => x.title.toLowerCase().includes(query) && x.href)
        .slice(0, 12);

      for (const match of matches) {
        const a = document.createElement('a');
        a.className = 'sr-search-result';
        a.href = match.href;
        a.innerHTML = `
          ${match.image ? `<img src="${match.image}" alt="">` : ''}
          <span>${match.title.replace(/</g, '&lt;')}</span>
        `;
        results.appendChild(a);
      }

      if (!matches.length) {
        const remote = document.createElement('a');
        remote.className = 'sr-search-result';
        remote.href = `/?s=${encodeURIComponent(input.value)}`;
        remote.innerHTML = `<span>Search SteamRIP for “${input.value.replace(/</g, '&lt;')}”</span>`;
        results.appendChild(remote);
      }
    });

    input.addEventListener('keydown', e => {
      if (e.key === 'Escape') closeSearch();
      if (e.key === 'Enter') {
        const q = cleanText(input.value);
        if (q) location.href = `/?s=${encodeURIComponent(q)}`;
      }
    });
  }

  function openSearch() {
    const overlay = document.querySelector('.sr-search');
    if (!overlay) return;
    overlay.classList.add('open');
    const input = overlay.querySelector('.sr-search-input');
    input.value = '';
    overlay.querySelector('.sr-search-results').innerHTML = '';
    setTimeout(() => input.focus(), 20);
  }

  function closeSearch() {
    document.querySelector('.sr-search')?.classList.remove('open');
  }

  function enhanceNav() {
    const nav = document.querySelector('#main-nav');
    if (!nav) return;

    nav.querySelectorAll('.search-compact-icon a, .tie-search-trigger, .tie-search-trigger-mobile')
      .forEach(a => {
        a.addEventListener('click', e => {
          e.preventDefault();
          openSearch();
        }, true);
      });

    buildMobileControls();
  }

  function enhanceHome() {
    const cards = document.querySelectorAll('.post-item');
    if (!cards.length) return;

    document.querySelectorAll('.post-item').forEach(item => {
      const title = item.querySelector('.the-post-title, .post-title');
      if (title) title.title = cleanText(title.textContent);
    });

    const sliders = document.querySelectorAll('.featured-area, .boxed-four-taller-slider, .tie-grid-slider');
    sliders.forEach(slider => {
      slider.style.borderRadius = '16px';
      slider.style.overflow = 'hidden';
    });
  }

  function enhanceListing() {
    const cards = document.querySelectorAll('.post-item');
    if (!cards.length) return;

    const firstHeading = document.querySelector('.mag-box-title h3, .block-title h3');
    if (firstHeading) {
      firstHeading.textContent = cleanText(firstHeading.textContent);
    }
  }

  function enhanceGamePage() {
    const article = document.querySelector('#the-post');
    if (!article) return;

    const content = article.querySelector('.entry-content');
    if (!content) return;

    const headings = [...content.querySelectorAll('h2, h3, h4')];
    headings.forEach(h => {
      if (!cleanText(h.textContent)) h.remove();
    });

    const images = [...content.querySelectorAll('img')];
    images.forEach(img => {
      const src = img.getAttribute('data-src') || img.getAttribute('data-lazy-src');
      if (src && (!img.src || img.src.startsWith('data:'))) {
        img.src = absolute(src);
      }
      img.loading = 'lazy';
    });
  }

  function addQuickSearchButton() {
    const components = document.querySelector('#main-nav .components');
    if (!components || components.querySelector('.sr-search-button')) return;

    const li = document.createElement('li');
    li.className = 'sr-search-button';
    li.innerHTML = `<a href="#" title="Search" aria-label="Search">⌕</a>`;
    li.querySelector('a').addEventListener('click', e => {
      e.preventDefault();
      openSearch();
    });
    components.prepend(li);
  }

  function init() {
    injectStyle();
    if (!document.body) return;

    addClasses();
    buildSearch();
    enhanceNav();
    addQuickSearchButton();

    if (isHome()) enhanceHome();
    if (isListing()) enhanceListing();
    if (isGamePage()) enhanceGamePage();

    const observer = new MutationObserver(() => {
      if (!document.body.classList.contains('sr-redesign-active')) {
        document.body.classList.add('sr-redesign-active');
      }
      if (!document.querySelector('.sr-search')) buildSearch();
    });

    observer.observe(document.body, { childList: true, subtree: true });

    window.addEventListener('keydown', e => {
      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        openSearch();
      }
    });
  }

  if (document.documentElement) injectStyle();
  ready(init);
})();

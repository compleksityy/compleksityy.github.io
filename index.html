// ==UserScript==
// @name        SteamRIP Reripped
// @namespace   bepi
// @icon        https://steamrip.com/wp-content/uploads/2021/06/cropped-favicon1-32x32.png
// @version     1.0
// @match       https://steamrip.com/*
// @grant       none
// @author      bypass
// @description
// ==/UserScript==

/*
    Thanks for using SteamRIP Reripped!
    Please keep in mind that this is meant to be a stupid little side-project
    and the code is absolutely spaghetti. But it works, and that's good enough
    for something this stupid.

    If you're going to fork this and make your own variation, please do supply
    SOME kind of credits. Please and thank you. :)

    List of changes:
    - Cleans up the background
    - New fonts and colors
    - Removed broken and unused links (like the Discord and Reddit icons)
    - Expands on the existing SteamRIP rounded edges update
    - Restyles a few elements to feel more modern
    - Changed font to Gesit Mono (Compleksity)

    ~ bepi
    ~ compleksity
*/

(() => {
  "use strict";

  async function steamRIP() {

    // Apply Stylesheet
    const reRippedStyle = `@import url('https://imbypass.pw/reripped/');`;
    const styleElement = document.createElement('style');
    styleElement.type = 'text/css';
    styleElement.textContent = reRippedStyle;
    (document.head || document.documentElement).appendChild(styleElement);

    // Apply Geist Mono
    const fontStyle = document.createElement('style');
    fontStyle.textContent = `
      @import url('https://fonts.googleapis.com/css2?family=Geist+Mono:wght@100..900&display=swap');

      :root {
        --font-header: 'Geist Mono', monospace !important;
        --font-body: 'Geist Mono', monospace !important;
      }

      body {
        font-family: 'Geist Mono', monospace !important;
      }

      p,
      a,
      li,
      label,
      input,
      textarea,
      button,
      select,
      option,
      h1,
      h2,
      h3,
      h4,
      h5,
      h6 {
        font-family: 'Geist Mono', monospace !important;
      }

      .post-title,
      .entry-title,
      .entry-content,
      .entry-content h1,
      .entry-content h2,
      .entry-content h3,
      .entry-content h4,
      .entry-content h5,
      .entry-content h6 {
        font-family: 'Geist Mono', monospace !important;
      }
    `;

    (document.head || document.documentElement).appendChild(fontStyle);

    // Update Favicon
    let links = document.querySelectorAll("link[rel*='icon']");

    if (links.length > 0) {
      links.forEach(link => {
        link.href = "https://cdn.discordapp.com/emojis/1529289995435643042.webp?size=64";
      });
    } else {
      let newLink = document.createElement('link');
      newLink.rel = 'icon';
      newLink.type = 'image/png';
      newLink.href = "https://cdn.discordapp.com/emojis/1529289995435643042.webp?size=64";
      document.head.appendChild(newLink);
    }

    // Update Page Title
    document.title = document.title.replace(" Â» ", " ");
    document.title = document.title.replace("Free Pre-installed Steam Games", "");
    document.title = document.title.replace("STEAMRIP", "SteamRIP");
    document.title = document.title.replace("Free Download ", " ");

    // Update SteamRIP Logo
    const logo = document.querySelector("img.tie-logo-img-default");
    const new_logo = document.querySelector("picture.tie-logo-default");
    const sticky_logo = document.querySelector("div#sticky-logo");

    if (logo) {
      logo.style = "display: none;";
    }

    if (new_logo) {
      new_logo.innerHTML = `
        <div style="
          color:white;
          font-size:32px;
          font-weight:900;
          font-family:'Geist Mono', monospace;
        ">
          <img
            style="margin-top:-12px;margin-right:6px;"
            src="https://cdn.discordapp.com/emojis/1529289995435643042.webp?size=32"
          />
          <span style="color:var(--accent);">STEAM</span>RIP
        </div>
      `;
    }

    if (sticky_logo) {
      sticky_logo.innerHTML = `
        <div style="
          color:white;
          font-size:32px;
          font-weight:900;
          font-family:'Geist Mono', monospace;
        ">
          <img
            style="margin-top:-12px;margin-right:6px;"
            src="https://cdn.discordapp.com/emojis/1529289995435643042.webp?size=32"
          />
          <span style="color:var(--accent);">STEAM</span>RIP
        </div>
      `;
    }

    const gameTitleElement = document.querySelector("h1.post-title.entry-title");
    const postTitleElement = document.querySelector("div.entry-content h2");

    if (!gameTitleElement || !postTitleElement) return;

    let gameName = gameTitleElement.textContent;
    let postTitle = postTitleElement.textContent;

    // Clean up game name & tags
    if (gameName.includes("Free Download")) {
      gameName = gameName.replace("Free Download ", "");
    }

    if (gameName.includes("+ Co-op")) {
      gameName = gameName.replace(" + Co-op", "");
      gameName += `
        <div style="
          color:var(--accent-elevated);
          font-size:21px;
          font-family:'Geist Mono', monospace;
        ">[Co-Op]</div>
      `;
    }

    if (gameName.includes("+ Online")) {
      gameName = gameName.replace(" + Online", "");
      gameName += `
        <div style="
          color:var(--accent-elevated);
          font-size:21px;
          font-family:'Geist Mono', monospace;
        ">[Online]</div>
      `;
    }

    if (gameName.includes("+ Multiplayer")) {
      gameName = gameName.replace(" + Multiplayer", "");
      gameName += `
        <div style="
          color:var(--accent-elevated);
          font-size:21px;
          font-family:'Geist Mono', monospace;
        ">[Multiplayer]</div>
      `;
    }

    gameTitleElement.innerHTML = gameName;

    // Clean up post title
    if (postTitle.includes("Direct Download")) {
      postTitle = postTitle.replace("Direct Download", "");
    }

    if (postTitle.includes("Direct Link")) {
      postTitle = postTitle.replace("Direct Link", "");
    }

    if (postTitle.includes("Directly Download")) {
      postTitle = postTitle.replace("Directly Download", "");
    }

    if (postTitle.includes("Free Download")) {
      postTitle = postTitle.replace("Free Download", "");
    }

    postTitleElement.innerText = postTitle;
  }

  steamRIP();

})();
installation
Tampermonkey is required.The script needs a userscript manager such as Tampermonkey to run in your browser.
  1. Use the copy code button above to copy the complete formatted script.
  2. Install Tampermonkey in your browser and make sure the extension is enabled.
  3. If your browser asks for permission to run user scripts, turn on Allow User Scripts for Tampermonkey.
  4. Open SteamRIP.com and open the Tampermonkey dashboard from your browser extensions.
  5. Create a new userscript, remove the starter template, and paste in the code you copied from this page.
  6. Save the script, return to SteamRIP.com, and refresh the page. The redesign should load automatically.
